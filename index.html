/** * MujaHeeBot - Penyimpanan progres siswa (Google Apps Script + Google Sheets) * * CARA PASANG (sekali saja, +- 5 menit): * 1. Buka sheets.new -> beri nama "Progres MujaHeeBot". * 2. Menu Ekstensi > Apps Script. Hapus isi default, tempel SELURUH kode ini. * 3. Ganti KODE_GURU di bawah dengan kode rahasia pilihan sekolah. * 4. Klik Terapkan (Deploy) > Deployment baru > jenis "Aplikasi web". * Jalankan sebagai : Saya (akun Anda) * Yang memiliki akses : Siapa saja * Klik Terapkan, izinkan akses, lalu SALIN "URL aplikasi web". * 5. Di index.html isi: const CLOUD_URL='URL-yang-tadi-disalin'; * (Jika kode ini diubah lagi, buat "Deployment baru" agar perubahan aktif.) */ const KODE_GURU = 'ganti-kode-ini'; const NAMA_SHEET = 'Progres'; const HEAD = ['id','nama','kelas','skor','selesai_total','selesai_kelas','mat','sns','agm','arb','cod','update']; function sheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let s = ss.getSheetByName(NAMA_SHEET); if (!s) { s = ss.insertSheet(NAMA_SHEET); s.appendRow(HEAD); s.setFrozenRows(1); } return s; } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function aman_(v) { // cegah rumus di Google Sheets v = String(v == null ? '' : v).slice(0, 80); return /^[=+\-@]/.test(v) ? "'" + v : v; } // Guru membaca data (wajib kode guru) function doGet(e) { const p = (e && e.parameter) || {}; if (p.code !== KODE_GURU) return json_({ error: 'kode salah' }); const v = sheet_().getDataRange().getValues(); v.shift(); return json_({ rows: v.map(function (r) { return { id: String(r[0]), name: String(r[1]), cls: String(r[2]), score: +r[3] || 0, done: +r[4] || 0, dg: +r[5] || 0, sub: { mat: +r[6] || 0, sns: +r[7] || 0, agm: +r[8] || 0, arb: +r[9] || 0, cod: +r[10] || 0 }, t: +r[11] || 0 }; }) }); } // Siswa mengirim progres / guru menghapus data function doPost(e) { let d; try { d = JSON.parse(e.postData.contents); } catch (x) { return json_({ error: 'bad request' }); } const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const s = sheet_(), last = s.getLastRow(); const ids = last > 1 ? s.getRange(2, 1, last - 1, 1).getValues().map(function (r) { return String(r[0]); }) : []; if (d.action === 'save' && d.rec) { const r = d.rec, sb = r.sub || {}; const row = [aman_(r.id), aman_(r.name), aman_(r.cls), +r.score || 0, +r.done || 0, +r.dg || 0, +sb.mat || 0, +sb.sns || 0, +sb.agm || 0, +sb.arb || 0, +sb.cod || 0, +r.t || Date.now()]; const i = ids.indexOf(String(r.id)); if (i < 0) s.appendRow(row); else s.getRange(i + 2, 1, 1, row.length).setValues([row]); } else if (d.action === 'delete' && d.code === KODE_GURU) { const i = ids.indexOf(String(d.id)); if (i >= 0) s.deleteRow(i + 2); } return json_({ ok: true }); } finally { lock.releaseLock(); } }
